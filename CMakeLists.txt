cmake_minimum_required(VERSION 3.20)
project(pathtracer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

find_package(OpenMP QUIET)
find_package(SDL2 QUIET)

if(SDL2_FOUND)
    message(STATUS "pathtracer: SDL2 found -> windowed display")
else()
    message(STATUS "pathtracer: SDL2 not found -> PPM frame output fallback")
endif()

function(add_tracer_target name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    if(MSVC)
        target_compile_options(${name} PRIVATE /O2 /W4)
    else()
        target_compile_options(${name} PRIVATE -O3 -Wall -Wextra)
    endif()
    if(MINGW)
        target_link_options(${name} PRIVATE -static)
    endif()
    if(OpenMP_CXX_FOUND)
        target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
    endif()
    if(SDL2_FOUND)
        target_link_libraries(${name} PRIVATE SDL2)
        target_compile_definitions(${name} PRIVATE PATHTRACER_HAS_SDL)
    endif()
endfunction()

add_tracer_target(pathtracer src/main.cpp)

# PPM region stats helper for the smoke check (see AGENTS.md); not a test.
add_tracer_target(smoke_stats tests/smoke_stats.cpp)

enable_testing()
foreach(t IN ITEMS vec3 box camera light tracer args)
    add_tracer_target(test_${t} tests/test_${t}.cpp)
    target_include_directories(test_${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
    add_test(NAME test_${t} COMMAND test_${t})
endforeach()
